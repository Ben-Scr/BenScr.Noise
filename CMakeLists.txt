cmake_minimum_required(VERSION 3.28)
project(BenScrNoise VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(BENSCR_NOISE_BUILD_TESTS "Build contract tests" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_NOISE_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_NOISE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(benscr_noise STATIC src/generator.cpp)
add_library(BenScr::Noise ALIAS benscr_noise)
set_target_properties(benscr_noise PROPERTIES EXPORT_NAME Noise CXX_EXTENSIONS OFF)
target_compile_features(benscr_noise PUBLIC cxx_std_20)
target_include_directories(benscr_noise PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(benscr_noise_checks target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BENSCR_NOISE_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported POSIX host.")
        endif()
    endif()
endfunction()
benscr_noise_checks(benscr_noise)

if(BUILD_TESTING AND BENSCR_NOISE_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(noise_contract_tests tests/noise_tests.cpp)
    target_link_libraries(noise_contract_tests PRIVATE BenScr::Noise Threads::Threads)
    benscr_noise_checks(noise_contract_tests)
    add_test(NAME noise_contract COMMAND noise_contract_tests)
    add_executable(noise_header_test tests/header.cpp)
    target_link_libraries(noise_header_test PRIVATE BenScr::Noise)
    benscr_noise_checks(noise_header_test)
    add_test(NAME standalone_header COMMAND noise_header_test)
endif()
if(BENSCR_NOISE_BUILD_EXAMPLES)
    add_executable(noise_example examples/noise.cpp)
    target_link_libraries(noise_example PRIVATE BenScr::Noise)
    benscr_noise_checks(noise_example)
endif()

install(TARGETS benscr_noise EXPORT BenScrNoiseTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/BenScrNoise)
configure_package_config_file(cmake/BenScrNoiseConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrNoiseConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrNoise)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BenScrNoiseConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BenScrNoiseConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrNoiseConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrNoise)
install(EXPORT BenScrNoiseTargets NAMESPACE BenScr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrNoise)
